═══════════════════════════════════════════════════════════════
   SHORTENN.ORG - QUICK START GUIDE (5 MINUTES)
═══════════════════════════════════════════════════════════════

📦 STEP 1: UPLOAD (2 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Login to Hostinger hPanel
2. Open File Manager
3. Navigate to public_html
4. Upload shortenn_laravel.zip
5. Right-click → Extract
6. Done! ✅

🗄️ STEP 2: CREATE DATABASE (1 minute)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: Databases → MySQL Databases
2. Click "Create New Database"
3. SAVE THESE:
   - Database Name: _____________
   - Username: _____________
   - Password: _____________
   - Host: localhost

📥 STEP 3: IMPORT DATABASE (30 seconds)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to phpMyAdmin
2. Select your database (left sidebar)
3. Click "Import" tab
4. Choose: database_export.sql
5. Click "Import" button
6. Wait for success ✅

⚙️ STEP 4: EDIT .ENV FILE (1 minute)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. File Manager → shortenn_laravel/.env
2. Right-click → Edit
3. UPDATE THESE 3 LINES:

   DB_DATABASE=YOUR_DATABASE_NAME_HERE
   DB_USERNAME=YOUR_DATABASE_USERNAME_HERE
   DB_PASSWORD=YOUR_DATABASE_PASSWORD_HERE

4. Save! ✅

🌐 STEP 5: POINT DOMAIN (30 seconds)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: Domains
2. Click on: shortenn.org
3. Change "Document Root" to:
   /public_html/shortenn_laravel/public
4. Save! ✅

🔒 STEP 6: SET PERMISSIONS (30 seconds)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Navigate to: shortenn_laravel/storage
2. Right-click → Permissions
3. Set: 755
4. Check: ✅ Apply to subdirectories
5. Click "Change"

6. Navigate to: shortenn_laravel/bootstrap/cache
7. Right-click → Permissions → Set: 755

✅ STEP 7: TEST (30 seconds)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Visit: https://shortenn.org
