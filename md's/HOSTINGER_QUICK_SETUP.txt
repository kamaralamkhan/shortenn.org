=====================================================
SHORTENN.ORG - HOSTINGER QUICK SETUP GUIDE
=====================================================

📋 STEP-BY-STEP DEPLOYMENT (5 Minutes)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 1: CREATE DATABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Login to Hostinger hPanel
2. Go to: Databases → MySQL Databases
3. Click "Create New Database"
4. Save credentials:
   ✅ Database Name: u123456789_shortenn
   ✅ Username: u123456789_shortenn_user
   ✅ Password: [your_password]
   ✅ Host: localhost

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 2: IMPORT DATABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: phpMyAdmin (in hPanel)
2. Select your database (left sidebar)
3. Click "Import" tab
4. Choose file: database_export.sql
5. Click "Go" button
6. Wait for success message ✅

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 3: UPLOAD PROJECT FILES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: File Manager
2. Navigate to: public_html
3. Upload ENTIRE shortenn_laravel folder
4. Wait for upload to complete

Skip these folders (not needed):
❌ node_modules/
❌ .git/

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 4: CONFIGURE .ENV FILE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. In File Manager, go to: shortenn_laravel/
2. Find: .env.example
3. Right-click → Copy → Rename to: .env
4. Right-click .env → Edit
5. Update these lines:

APP_NAME=Shortenn
APP_ENV=production
APP_DEBUG=false
APP_URL=https://shortenn.org

DB_CONNECTION=mysql
DB_HOST=localhost
DB_PORT=3306
DB_DATABASE=u123456789_shortenn          ← YOUR DATABASE NAME
DB_USERNAME=u123456789_shortenn_user     ← YOUR USERNAME
DB_PASSWORD=your_password_here           ← YOUR PASSWORD

6. Save changes

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 5: GENERATE APP KEY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: https://generate-random.org/laravel-key-generator
2. Copy the generated key (looks like: base64:ABC123...)
3. Edit .env file
4. Add: APP_KEY=base64:ABC123...
5. Save

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 6: POINT DOMAIN TO PUBLIC FOLDER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Go to: Domains (in hPanel)
2. Click on: shortenn.org
3. Find: "Document Root" or "Website Root"
4. Change to: /public_html/shortenn_laravel/public
5. Save changes

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 7: SET FILE PERMISSIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
In File Manager:

1. Go to: shortenn_laravel/storage/
   Right-click → Permissions → Set to: 755
   Check: Apply to subdirectories
   
2. Go to: shortenn_laravel/bootstrap/cache/
   Right-click → Permissions → Set to: 755

🔧 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Problem: 500 Internal Server Error
Solution:
✅ Check .env file is configured correctly
✅ Check storage/ permissions are 755
✅ Check bootstrap/cache/ permissions are 755

Problem: Database Connection Error
Solution:
✅ Verify database name, username, password in .env
✅ Make sure database was imported successfully

Problem: Blank Page
Solution:
✅ Set APP_DEBUG=true in .env (temporarily)
✅ Check error logs in storage/logs/

Problem: CSS/JS Not Loading
Solution:
✅ Check domain points to /public folder
✅ Clear browser cache
✅ Check .htaccess file exists in public/
